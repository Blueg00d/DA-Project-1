143/*
144 * Auxiliary function to remove an outgoing edge (with a given destination (d))
145 * from a vertex (this).
146 * Returns true if successful, and false if such edge does not exist.
147 */
148template <class T>
149bool Vertex<T>::removeEdge(T in) {
150 bool removedEdge = false;
151 auto it = adj.begin();
152 while (it != adj.end()) {
153 Edge<T> *edge = *it;
154 Vertex<T> *dest = edge->getDest();
155 if (dest->getInfo() == in) {
156 it = adj.erase(it);
157 deleteEdge(edge);
158 removedEdge = true; // allows for multiple edges to connect the same pair of vertices (multigraph)
159 }
160 else {
161 it++;
162 }
163 }
164 return removedEdge;
165}

167/*
168 * Auxiliary function to remove an outgoing edge of a vertex.
169 */
170template <class T>
171void Vertex<T>::removeOutgoingEdges() {
172 auto it = adj.begin();
173 while (it != adj.end()) {
174 Edge<T> *edge = *it;
175 it = adj.erase(it);
176 deleteEdge(edge);
177 }
178}

222 Vertex<T> *dest = edge->getDest();
223 // Remove the corresponding edge from the incoming list

230 it++;

312 * Adds a vertex with a given content or info (in) to a graph (this).
313 * Returns true if successful, and false if a vertex with that content already exists.

315template <class T>
316bool Graph<T>::addVertex(const T &in) {
317 if (findVertex(in) != nullptr)
318 return false;
319 auto newVertex = new Vertex<T>(in);
320 vertexSet.push_back(newVertex);
321 vertexMap[in] = newVertex; // O(1) map insertion
322 return true;
323}

325/*
326 * Removes a vertex with a given content (in) from a graph (this), and
327 * all outgoing and incoming edges.
328 * Returns true if successful, and false if such vertex does not exist.
329 */
330template <class T>
331bool Graph<T>::removeVertex(const T &in) {
332 for (auto it = vertexSet.begin(); it != vertexSet.end(); it++) {
333 if ((*it)->getInfo() == in) {
334 auto v = *it;
335 v->removeOutgoingEdges();
336 for (auto u : vertexSet) {
337 u->removeEdge(v->getInfo());
338 }
339 vertexSet.erase(it);
340 vertexMap.erase(in); // O(1) map deletion
341 delete v;
342 return true;
343 }
344 }
345 return false;
346}

382 return false;

390// Function to perform DFS and find an augmenting path
391template <class T>
392bool Graph<T>::dfsFindAugmentingPath(Vertex<T>* v, Vertex<T>* t, double& flow) {
393 if (v == t) return true;
394 v->setVisited(true);
395
396 // Forward edges
397 for (auto e : v->getAdj()) {
398
399 Vertex<T>* w = e->getDest();
400 double residual = e->getWeight() - e->getFlow();
401 if (!w->isVisited() && residual > 0) {
402 w->setPath(e);
403 if (dfsFindAugmentingPath(w, t, flow)) {
404 //Keep edge to backtrack
405 flow = std::min(flow, residual);
406 return true;
407 }
408 }
409 }
410
411 // Backward edges
412 for (auto e : v->getIncoming()) {
413 double residual = e->getFlow();
414 Vertex<T>* w = e->getOrig();
415
416 if (!w->isVisited() && residual > 0) {
417 w->setPath(e); //keeps edge; destination for backtracking
418 if (dfsFindAugmentingPath(w, t, flow)) {
419 flow = std::min(flow, residual);
420
421 return true;
422 }
423 }
424 }
425 return false;
426}

428// Ford-Fulkerson algorithm (change has return type int)
429template <class T>
430double Graph<T>::fordFulkerson(int source, int target) {
431 Vertex<T>* s = findVertex(source);
432 Vertex<T>* t = findVertex(target);
433
434 if (s == nullptr || t == nullptr || s == t)
435 throw std::logic_error("Invalid source and/or target vertex");
436
437 double maxFlow = 0;
438
439 // Initialize all flows to 0
440 for (auto v : vertexSet) {
441 for (auto e : v->getAdj()) {
442 e->setFlow(0);
443 }
444 }
445 double flow=INF;
446 for (Vertex<T>* v : vertexSet)
447 {
448 v->setVisited(false);
449 v->setPath(nullptr);
450 }
451 while (dfsFindAugmentingPath(s, t, flow))
452 {
453 Vertex<T>* vertex = t;
454 maxFlow += flow;
455
456 for (Vertex<T>* v = t; v != s; )
457 {
458 Edge<T>* e = v->getPath();
459 double floww = e->getFlow();
460
461 if (e->getDest() == v) //If forward edge
462 {
463 e->setFlow(floww + flow);
464 v = e->getOrig();
465 } else //Backward edge
466 {
467 e->setFlow(floww - flow);
468 v = e->getDest();
469 }
470 }
471
472 flow = INF;
473
474 for (Vertex<T>* v : vertexSet)
475 {
476 v->setVisited(false);
477 v->setPath(nullptr);
478 }
479
480 }
481 return maxFlow;
482}
