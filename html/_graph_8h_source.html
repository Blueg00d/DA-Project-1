316/*
317 * Auxiliary function to remove an outgoing edge (with a given destination (d))
318 * from a vertex (this).
319 * Returns true if successful, and false if such edge does not exist.
320 */
321template <class T>
322bool Vertex<T>::removeEdge(T in) {
323 bool removedEdge = false;
324 auto it = adj.begin();
325 while (it != adj.end()) {
326 Edge<T> *edge = *it;
327 Vertex<T> *dest = edge->getDest();
328 if (dest->getInfo() == in) {
329 it = adj.erase(it);
330 deleteEdge(edge);
331 removedEdge = true; // allows for multiple edges to connect the same pair of vertices (multigraph)
332 }
333 else {
334 it++;
335 }
336 }
337 return removedEdge;
338}

347 Edge<T> *edge = *it;

350 }

393template <class T>
394void Vertex<T>::deleteEdge(Edge<T> *edge) {
395 Vertex<T> *dest = edge->getDest();
396 // Remove the corresponding edge from the incoming list
397 auto it = dest->incoming.begin();
398 while (it != dest->incoming.end()) {
399 if ((*it)->getOrig()->getInfo() == info) {
400 it = dest->incoming.erase(it);
401 }
402 else {
403 it++;
404 }

499/*
500 * Removes a vertex with a given content (in) from a graph (this), and
501 * all outgoing and incoming edges.
502 * Returns true if successful, and false if such vertex does not exist.
503 */
504template <class T>
505bool Graph<T>::removeVertex(const T &in) {
506 for (auto it = vertexSet.begin(); it != vertexSet.end(); it++) {
507 if ((*it)->getInfo() == in) {
508 auto v = *it;
509 v->removeOutgoingEdges();
510 for (auto u : vertexSet) {
511 u->removeEdge(v->getInfo());
512 }
513 vertexSet.erase(it);
514 vertexMap.erase(in); // O(1) map deletion
515 delete v;
516 return true;
517 }
518 }
519 return false;
520}

552bool Graph<T>::addBidirectionalEdge(const T &sourc, const T &dest, double w) {
553 auto v1 = findVertex(sourc);
554 auto v2 = findVertex(dest);

556 return false;

559 e1->setReverse(e2);
560 e2->setReverse(e1);

564// Function to perform DFS and find an augmenting path
565template <class T>
566bool Graph<T>::dfsFindAugmentingPath(Vertex<T>* v, Vertex<T>* t, double& flow) {
567 if (v == t) return true;
568 v->setVisited(true);
569
570 // Forward edges
571 for (auto e : v->getAdj()) {
572
573 Vertex<T>* w = e->getDest();
574 double residual = e->getWeight() - e->getFlow();
575 if (!w->isVisited() && residual > 0) {
576 w->setPath(e);
577 if (dfsFindAugmentingPath(w, t, flow)) {
578 //Keep edge to backtrack
579 flow = std::min(flow, residual);
580 return true;
581 }
582 }
583 }
584
585 // Backward edges
586 for (auto e : v->getIncoming()) {
587 double residual = e->getFlow();
588 Vertex<T>* w = e->getOrig();
589
590 if (!w->isVisited() && residual > 0) {
591 w->setPath(e); //keeps edge; destination for backtracking
592 if (dfsFindAugmentingPath(w, t, flow)) {
593 flow = std::min(flow, residual);
594
595 return true;
596 }
597 }
598 }
599 return false;
600}

602// Ford-Fulkerson algorithm
603template <class T>
604double Graph<T>::fordFulkerson(int source, int target) {
605 Vertex<T>* s = findVertex(source);
606 Vertex<T>* t = findVertex(target);
607
608 if (s == nullptr || t == nullptr || s == t)
609 throw std::logic_error("Invalid source and/or target vertex");
610
611 double maxFlow = 0;
612
613 // Initialize all flows to 0
614 for (auto v : vertexSet) {
615 for (auto e : v->getAdj()) {
616 e->setFlow(0);
617 }
618 }
619 double flow=INF;
620 for (Vertex<T>* v : vertexSet)
621 {
622 v->setVisited(false);
623 v->setPath(nullptr);
624 }
625 while (dfsFindAugmentingPath(s, t, flow))
626 {
627 Vertex<T>* vertex = t;
628 maxFlow += flow;
629
630 for (Vertex<T>* v = t; v != s; )
631 {
632 Edge<T>* e = v->getPath();
633 double floww = e->getFlow();
634
635 if (e->getDest() == v) //If forward edge
636 {
637 e->setFlow(floww + flow);
638 v = e->getOrig();
639 } else //Backward edge
640 {
641 e->setFlow(floww - flow);
642 v = e->getDest();
643 }
644 }
645
646 flow = INF;
647
648 for (Vertex<T>* v : vertexSet)
649 {
650 v->setVisited(false);
651 v->setPath(nullptr);
652 }
653
654 }
655 return maxFlow;
656}

658template <class T>
659double Graph<T>::resumeFordFulkerson(int source, int target, double currentFlow) {
660 Vertex<T>* s = findVertex(source);
661 Vertex<T>* t = findVertex(target);
662
663 if (s == nullptr || t == nullptr || s == t)
664 throw std::logic_error("Invalid source and/or target vertex");
665
666 double maxFlow = currentFlow;
667 double flow = INF;
668
669 for (Vertex<T>* v : vertexSet) {
670 v->setVisited(false);
671 v->setPath(nullptr);
672 }
673
674 while (dfsFindAugmentingPath(s, t, flow)) {
675 Vertex<T>* vertex = t;
676 maxFlow += flow;
677
678 for (Vertex<T>* v = t; v != s; ) {
679 Edge<T>* e = v->getPath();
